   102	async def _broadcast(document: dict[str, Any]) -> None:
   103	    stale: list[WebSocket] = []
   104	    for client in tuple(live_clients):
   105	        try:
   106	            await client.send_json(document)
   107	        except Exception:
   108	            stale.append(client)
   109	    for client in stale:
   110	        live_clients.discard(client)

   113	def _store(request: Request) -> tuple[deque[dict[str, Any]], set[str]]:
   114	    if not hasattr(request.app.state, "decision_alerts"):
   115	        request.app.state.decision_alerts = deque(maxlen=1000)
   116	        request.app.state.decision_alert_keys = set()
   117	    return request.app.state.decision_alerts, request.app.state.decision_alert_keys
   118	
   119	
   120	def _trip_store(request: Request) -> dict[str, dict[str, Any]]:
   121	    if not hasattr(request.app.state, "live_trip_sessions"):
   122	        request.app.state.live_trip_sessions = {}
   123	    return request.app.state.live_trip_sessions

   126	def _ensure_trip(
   127	    request: Request,
   128	    trip_id: str,
   129	    metadata: dict[str, Any] | None = None,
   130	) -> dict[str, Any]:
   131	    sessions = _trip_store(request)
   132	    session = sessions.setdefault(
   133	        trip_id,
   134	        {
   135	            "trip_id": trip_id,
   136	            "status": "pending",
   137	            "metadata": {},
   138	            "latest_snapshot": None,
   139	            "snapshot_history": deque(maxlen=2400),
   140	        },
   141	    )
   142	    if metadata:
   143	        session["metadata"] = {**session["metadata"], **metadata}
   144	    return session

   154	@router.post("", status_code=status.HTTP_202_ACCEPTED)
   155	async def receive_alert(
   156	    payload: DecisionEventPayload,
   157	    request: Request,
   158	    idempotency_key: str = Header(alias="Idempotency-Key"),
   159	) -> dict[str, Any]:
   160	    if idempotency_key != payload.idempotency_key:
   161	        raise HTTPException(status_code=400, detail="Idempotency-Key mismatch")
   162	    alerts, keys = _store(request)
   163	    duplicate = idempotency_key in keys
   164	    if not duplicate:
   165	        document = payload.model_dump(mode="json")
   166	        _ensure_trip(request, payload.trip_id)
   167	        alerts.append(document)
   168	        keys.add(idempotency_key)
   169	        await _broadcast(document)
   170	        publisher = getattr(request.app.state, "carsky_publisher", None)
   171	        if publisher is not None and "driver_display" in payload.audiences:
   172	            await publisher.enqueue(
   173	                carsky_mapper.map_decision_event(document),
   174	                dedup_key=payload.idempotency_key,
   175	                kind=DeliveryKind.TRANSITION,
   176	            )
   177	    return {
   178	        "accepted": True,
   179	        "duplicate": duplicate,
   180	        "event_id": payload.event_id,
   181	    }

   192	@router.post("/trips/register", status_code=status.HTTP_202_ACCEPTED)
   193	async def register_live_trips(
   194	    payload: TripRegistrationPayload,
   195	    request: Request,
   196	) -> dict[str, Any]:
   197	    sessions = _trip_store(request)
   198	    if payload.reset_existing:
   199	        reset_ids = {trip.trip_id for trip in payload.trips}
   200	        for trip_id in reset_ids:
   201	            sessions.pop(trip_id, None)
   202	        for attribute in ("latest_cabin_frames", "latest_road_frames"):
   203	            frames = getattr(request.app.state, attribute, {})
   204	            for trip_id in reset_ids:
   205	                frames.pop(trip_id, None)
   206	        alerts, keys = _store(request)
   207	        retained = [
   208	            item for item in alerts if item.get("trip_id") not in reset_ids
   209	        ]
   210	        alerts.clear()
   211	        alerts.extend(retained)
   212	        keys.clear()
   213	        keys.update(item["idempotency_key"] for item in retained)
   214	    for trip in payload.trips:
   215	        _ensure_trip(request, trip.trip_id, trip.metadata)
   216	    return {
   217	        "accepted": True,
   218	        "count": len(payload.trips),
   219	        "trip_ids": [trip.trip_id for trip in payload.trips],
   220	    }
