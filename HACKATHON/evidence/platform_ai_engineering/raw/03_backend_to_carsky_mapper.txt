   129	            "harsh_accel": "HARSH_ACCEL",
   130	            "harsh_corner": "HARSH_CORNER",
   131	        }.get(alert_type, "NONE")
   132	        action_code = {
   133	            "collision_risk": "BRAKE_SAFE",
   134	            "microsleep": "TAKE_BREAK",
   135	            "driver_drowsiness": "TAKE_BREAK",

   158	    def map_live_snapshot(
   159	        self,
   160	        snapshot: Mapping[str, Any],
   161	        *,
   162	        data_age_ms: int = 0,
   163	        transition: str | None = None,
   164	        realtime: bool = True,
   165	    ) -> dict[str, list[dict[str, Any]]]:
   166	        """Map live dashboard telemetry to the BTC-compatible speed-mux VHAL path.
   167	
   168	        Current CarSky/AAOS only accepts standard Vehicle.Speed. The APK V2.2
   169	        decodes decimal groups 41..50 from this one property.
   170	        """
   171	        if data_age_ms < 0:
   172	            raise ValueError("data_age_ms must not be negative")
   173	
   174	        risk = self._number(snapshot.get("risk_score"), None)
   175	        ttc = self._number(snapshot.get("predicted_ttc_sec"), None)
   176	        driver_state = str(snapshot.get("driver_state") or "alert").lower()
   177	        alertness = self._number(snapshot.get("alertness_score"), None)
   178	        speed = self._number(snapshot.get("speed_kmh"), None)
   179	        safe_score = self._number(snapshot.get("safe_driving_score"), None)
   180	
   181	        severity = str(snapshot.get("severity") or "").upper()
   182	        if severity not in {"SAFE", "WARNING", "CRITICAL", "RECOVERY"}:
   183	            severity = self._derive_severity(risk or 0.0, ttc, driver_state)
   184	
   185	        action = str(snapshot.get("recommended_action") or "")
   186	        if action not in {"NONE", "FOCUS_FORWARD", "TAKE_BREAK", "BRAKE_SAFE", "REDUCE_SPEED"}:
   187	            action = self._derive_action(snapshot, ttc, driver_state, severity)
   188	
   189	        values: list[float] = [
   190	            self._mux(42, self._severity_code(severity)),
   191	            self._mux(43, self._driver_state_code(driver_state)),
   192	            self._mux(46, 1 if severity == "CRITICAL" or bool(snapshot.get("critical_alert")) else 0),
   193	            self._mux(47, self._ai_status_code(str(snapshot.get("ai_status") or "ONLINE"))),
   194	            self._mux(48, self._action_code(action)),
   195	        ]
   196	        if risk is not None and math.isfinite(risk):
   197	            values.insert(0, self._mux(41, risk))
   198	        if alertness is not None and math.isfinite(alertness):
   199	            values.append(self._mux(44, self._scale(alertness, 100)))
   200	        if ttc is not None and math.isfinite(ttc):
   201	            values.append(self._mux(45, self._scale(ttc, 10)))
   202	        if speed is not None and math.isfinite(speed):
   203	            values.append(self._mux(49, speed))
   204	        if safe_score is not None and math.isfinite(safe_score):
   205	            values.append(self._mux(50, safe_score))
   206	
   207	        signals = [{"path": self.paths.speed_mux, "value": value} for value in values]
   208	        return {
   209	            "transport": "vehicle-speed-mux",
   210	            "transition": transition,
   211	            "realtime": realtime,
   212	            "signals": signals,
   213	        }

   215	    @staticmethod
   216	    def _number(value: Any, default: float | None) -> float | None:
   217	        if value is None:
   218	            return default
   219	        try:
   220	            parsed = float(value)
   221	        except (TypeError, ValueError):
   222	            return default
   223	        return parsed if math.isfinite(parsed) else default

   225	    @staticmethod
   226	    def _scale(value: float | None, scale: float) -> float:
   227	        if value is None:
   228	            return 0.0
   229	        return value * scale

   231	    @staticmethod
   232	    def _mux(group: int, payload: float) -> float:
   233	        clamped = max(0.0, min(999.0, float(payload)))
   234	        return group + (clamped / 1000.0)

   236	    @staticmethod
   237	    def _derive_severity(risk: float, ttc: float | None, driver_state: str) -> str:
   238	        if risk >= 75 or (ttc is not None and ttc <= 1.5) or driver_state == "microsleep":
   239	            return "CRITICAL"
   240	        if risk >= 40 or (ttc is not None and ttc <= 3.0) or driver_state in {"drowsy", "yawning", "distracted"}:
   241	            return "WARNING"
   242	        return "SAFE"
